import java.util.Scanner;
import java.util.Random;

public class BarangayPatrol {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        Random rand = new Random();
        int[] tank = {80, 80, 80};
        int day = 1;

        System.out.println("=== BARANGAY TANK PATROL ===");
        System.out.println("Goal: Survive 15 days. Supply is 20% per day.");
        System.out.println("!!!Good luck!!!");

        while (day <= 15) {
            System.out.println("========== DAY " + day + " / 15 ==========");
            System.out.println("Tank A: " + tank[0] + "% | Tank B: " + tank[1] + "% | Tank C: " + tank[2] + "%");

            if (tank[0] < 40 || tank[1] < 40 || tank[2] < 40) {
                System.out.println("!!! WARNING: Low level! Need refill!");
            }

            System.out.print("Where to add 20%? (A/B/C): ");
            String choice = sc.next().toUpperCase();
            int idx = -1;
            if (choice.equals("A")) idx = 0;
            else if (choice.equals("B")) idx = 1;
            else if (choice.equals("C")) idx = 2;

            if (idx != -1) {
                tank[idx] += 20;
                if (tank[idx] > 100) tank[idx] = 100;
                System.out.println("> Added 20% to Tank " + choice);
            } else {
                System.out.println("> Invalid! No supply added.");
            }

            int dailyBawas;
            if (day <= 5) dailyBawas = 6;  
            else if (day <= 10) dailyBawas = 8; 
            else dailyBawas = 10;              

            System.out.println("> Daily barangay usage: -" + dailyBawas + "% all tanks");
            for (int i = 0; i < 3; i++) tank[i] -= dailyBawas;

            int bawas1 = rand.nextInt(5) + 8; // 8,9,10,11,12
            int target1 = rand.nextInt(3);

            System.out.println("> Leak: Tank " + (char)('A'+target1) + " -" + bawas1 + "%");
            tank[target1] -= bawas1;

            for (int i = 0; i < 3; i++) {
                if (tank[i] <= 0) {
                    System.out.println("=========================");
                    System.out.println("Tank " + (char)('A'+i) + " empty! Game Over at Day " + day);
                    if (day <= 6) {
                        System.out.println("Lesson: Wag mag focus sa isang tank lang!");
                    } else if (day <= 10) {
                        System.out.println("Lesson: Nako malapit ka na sana! Lagi mo lang tignan sino pinakamababa sa kanila.");
                    } else {
                        System.out.println("Lesson: Sayang! Day " + day + " ka na! Konti na lang!");
                        System.out.println("Technique: Wag mag-100% sa isa, keep all above 35%!");
                    }
                    System.out.println("=========================");
                    
                    return;
                }
            }

            System.out.println("> Remaining: A=" + tank[0] + "% B=" + tank[1] + "% C=" + tank[2] + "%");
            day++;
        }

        System.out.println("==============================");
        System.out.println("YOU WIN! YOU SURVIVED 15 DAYS!");
        System.out.println("Final: A=" + tank[0] + "% B=" + tank[1] + "% C=" + tank[2] + "%");
        System.out.println("Ang galing! Na-gets mo na technique!");
        System.out.println("Mahirap sa una, pero nung naintindihan mo, nanalo ka! Good work!!!!");
        System.out.println("==============================");
    }
}
